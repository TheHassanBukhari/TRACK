-- =========================================
-- TRACK DATABASE TABLES
-- =========================================

-- OFFICE
CREATE TABLE OFFICE (
    OFFICE_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    ROOM_NO VARCHAR2(20)
);

-- OFFICER
CREATE TABLE OFFICER (
    OFFICER_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    FIRST_NAME VARCHAR2(50) NOT NULL,
    LAST_NAME VARCHAR2(50) NOT NULL,
    HIRE_DATE DATE,
    ROLE VARCHAR2(50),
    RANK VARCHAR2(50),
    GENDER VARCHAR2(10),
    
    CONSTRAINT CHK_OFFICER_GENDER
    CHECK (GENDER IN ('Male', 'Female'))
);

-- OFFICER_PHONE
CREATE TABLE OFFICER_PHONE (
    OFFICER_ID NUMBER,
    PHONE_NO VARCHAR2(20),

    PRIMARY KEY (OFFICER_ID, PHONE_NO),

    CONSTRAINT FK_OFFICER_PHONE
    FOREIGN KEY (OFFICER_ID)
    REFERENCES OFFICER(OFFICER_ID)
    ON DELETE CASCADE
);

-- ADMIN_OFFICER
CREATE TABLE ADMIN_OFFICER (
    OFFICER_ID NUMBER PRIMARY KEY,
    ADMIN_LEVEL VARCHAR2(30),
    DEPARTMENT_CONTROL VARCHAR2(50),

    CONSTRAINT FK_ADMIN_OFFICER
    FOREIGN KEY (OFFICER_ID)
    REFERENCES OFFICER(OFFICER_ID)
    ON DELETE CASCADE
);

-- REGULAR_OFFICER
CREATE TABLE REGULAR_OFFICER (
    OFFICER_ID NUMBER PRIMARY KEY,
    SHIFT_TIMING VARCHAR2(50),
    PATROL_AREA VARCHAR2(100),

    CONSTRAINT FK_REGULAR_OFFICER
    FOREIGN KEY (OFFICER_ID)
    REFERENCES OFFICER(OFFICER_ID)
    ON DELETE CASCADE
);

-- ADD FK IN OFFICE AFTER ADMIN_OFFICER EXISTS
ALTER TABLE OFFICE
ADD OFFICER_ID NUMBER UNIQUE;

ALTER TABLE OFFICE
ADD CONSTRAINT FK_OFFICE_ADMIN
FOREIGN KEY (OFFICER_ID)
REFERENCES ADMIN_OFFICER(OFFICER_ID);

-- FIR
CREATE TABLE FIR (
    FIR_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    FIR_NO VARCHAR2(30) UNIQUE,
    FIR_DATE DATE,
    DESCR VARCHAR2(500),
    OFFICER_ID NUMBER,

    CONSTRAINT FK_FIR_OFFICER
    FOREIGN KEY (OFFICER_ID)
    REFERENCES OFFICER(OFFICER_ID)
);

-- CASES
CREATE TABLE CASES (
    CASE_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    C_TITLE VARCHAR2(100),
    OPEN_DATE DATE,
    CLOSE_DATE DATE,
    C_STATUS VARCHAR2(30),
    FIR_ID NUMBER UNIQUE,

    CONSTRAINT FK_CASE_FIR
    FOREIGN KEY (FIR_ID)
    REFERENCES FIR(FIR_ID),

    CONSTRAINT CHK_CASE_STATUS
    CHECK (C_STATUS IN ('Open', 'Closed', 'Under Investigation'))
);

-- EVIDENCE
CREATE TABLE EVIDENCE (
    CASE_ID NUMBER,
    EVID_ID NUMBER,
    E_DESC VARCHAR2(300),
    COLLECTED_DATE DATE,
    EVID_TYPE VARCHAR2(50),

    PRIMARY KEY (CASE_ID, EVID_ID),

    CONSTRAINT FK_EVIDENCE_CASE
    FOREIGN KEY (CASE_ID)
    REFERENCES CASES(CASE_ID)
    ON DELETE CASCADE
);

-- CRIME
CREATE TABLE CRIME (
    CRIME_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    CRIME_TYPE VARCHAR2(50),
    C_DATE DATE,
    C_LOCATION VARCHAR2(100),
    C_DESCR VARCHAR2(500)
);

-- VICTIMS
CREATE TABLE VICTIMS (
    VICTIM_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    FIRST_NAME VARCHAR2(50),
    LAST_NAME VARCHAR2(50),
    GENDER VARCHAR2(10),

    CONSTRAINT CHK_VICTIM_GENDER
    CHECK (GENDER IN ('Male', 'Female'))
);

-- VICTIM_CONTACT
CREATE TABLE VICTIM_CONTACT (
    VICTIM_ID NUMBER,
    CONTACT_NO VARCHAR2(20),

    PRIMARY KEY (VICTIM_ID, CONTACT_NO),

    CONSTRAINT FK_VICTIM_CONTACT
    FOREIGN KEY (VICTIM_ID)
    REFERENCES VICTIMS(VICTIM_ID)
    ON DELETE CASCADE
);

-- CRIMINAL
CREATE TABLE CRIMINAL (
    CR_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    FIRST_NAME VARCHAR2(50),
    LAST_NAME VARCHAR2(50),
    STREET VARCHAR2(100),
    DISTRICT VARCHAR2(100),
    CITY VARCHAR2(100),
    STATUS VARCHAR2(30),
    DOB DATE,
    GENDER VARCHAR2(10),

    CONSTRAINT CHK_CRIMINAL_GENDER
    CHECK (GENDER IN ('Male', 'Female'))
);

-- CRIMINAL_RECORD
CREATE TABLE CRIMINAL_RECORD (
    CR_ID NUMBER PRIMARY KEY,
    RECORD_ID NUMBER UNIQUE,
    JAIL_HISTORY VARCHAR2(500),
    PREV_CASES NUMBER,

    CONSTRAINT FK_CRIMINAL_RECORD
    FOREIGN KEY (CR_ID)
    REFERENCES CRIMINAL(CR_ID)
    ON DELETE CASCADE
);

-- INVESTIGATES
CREATE TABLE INVESTIGATES (
    OFFICER_ID NUMBER,
    CRIME_ID NUMBER,

    PRIMARY KEY (OFFICER_ID, CRIME_ID),

    CONSTRAINT FK_INVESTIGATES_OFFICER
    FOREIGN KEY (OFFICER_ID)
    REFERENCES OFFICER(OFFICER_ID)
    ON DELETE CASCADE,

    CONSTRAINT FK_INVESTIGATES_CRIME
    FOREIGN KEY (CRIME_ID)
    REFERENCES CRIME(CRIME_ID)
    ON DELETE CASCADE
);

-- BELONGS_TO
CREATE TABLE BELONGS_TO (
    CASE_ID NUMBER,
    CRIME_ID NUMBER,

    PRIMARY KEY (CASE_ID, CRIME_ID),

    CONSTRAINT FK_BELONGS_CASE
    FOREIGN KEY (CASE_ID)
    REFERENCES CASES(CASE_ID)
    ON DELETE CASCADE,

    CONSTRAINT FK_BELONGS_CRIME
    FOREIGN KEY (CRIME_ID)
    REFERENCES CRIME(CRIME_ID)
    ON DELETE CASCADE
);

-- INVOLVES
CREATE TABLE INVOLVES (
    CRIME_ID NUMBER,
    VICTIM_ID NUMBER,

    PRIMARY KEY (CRIME_ID, VICTIM_ID),

    CONSTRAINT FK_INVOLVES_CRIME
    FOREIGN KEY (CRIME_ID)
    REFERENCES CRIME(CRIME_ID)
    ON DELETE CASCADE,

    CONSTRAINT FK_INVOLVES_VICTIM
    FOREIGN KEY (VICTIM_ID)
    REFERENCES VICTIMS(VICTIM_ID)
    ON DELETE CASCADE
);

-- COMMITS
CREATE TABLE COMMITS (
    CR_ID NUMBER,
    CRIME_ID NUMBER,

    PRIMARY KEY (CR_ID, CRIME_ID),

    CONSTRAINT FK_COMMITS_CRIMINAL
    FOREIGN KEY (CR_ID)
    REFERENCES CRIMINAL(CR_ID)
    ON DELETE CASCADE,

    CONSTRAINT FK_COMMITS_CRIME
    FOREIGN KEY (CRIME_ID)
    REFERENCES CRIME(CRIME_ID)
    ON DELETE CASCADE
);
