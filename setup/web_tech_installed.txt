### 2. Installed Software & Dependencies

```bash
# Node.js
curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -
sudo apt install -y nodejs

# Verify
node --version
```

```bash
# Project initialization
cd ~/Desktop/TRACK
mkdir webapp
cd webapp
npm init -y
```

```bash
# NPM packages
npm install express mongodb oracledb cors
```

```bash
# Oracle Instant Client (for oracledb to work)
cd ~/Downloads
wget https://download.oracle.com/otn_software/linux/instantclient/218000/instantclient-basic-linux.x64-21.8.0.0.0dbru.zip
unzip instantclient-basic-linux.x64-21.8.0.0.0dbru.zip
sudo mkdir -p /opt/oracle
sudo mv instantclient_21_8 /opt/oracle/
echo 'export LD_LIBRARY_PATH=/opt/oracle/instantclient_21_8:$LD_LIBRARY_PATH' >> ~/.bashrc
source ~/.bashrc
```
